We check read & write access first. Once linked, every logged set and workout setup change saves to your Sheet automatically.

How to enable write access (30s)
  1. In your Google Sheet, open Extensions → Apps Script.
  2. Paste the copied script and click Deploy → New deployment.
  3. Choose Web app (Access: Anyone) and paste the Web App URL below.
function doGet(e) {
  var ss = e.parameter.sheetUrl ? SpreadsheetApp.openByUrl(e.parameter.sheetUrl) : SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var rows = sheet.getDataRange().getValues();
  var isEmpty = (rows.length <= 1 && String(rows[0][0] || '').trim() === '');
  var metaSheet = ss.getSheetByName('KINETIC_DATA');
  var metaJson = metaSheet ? String(metaSheet.getRange('A1').getValue() || '').trim() : '';
  return ContentService.createTextOutput(JSON.stringify({
    ok: true, canWrite: true, isEmpty: isEmpty,
    hasKineticData: metaJson.indexOf('"kineticVersion"') !== -1,
    kineticJson: metaJson || null, rowCount: rows.length
  })).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  var ss = payload.sheetUrl ? SpreadsheetApp.openByUrl(payload.sheetUrl) : SpreadsheetApp.getActiveSpreadsheet();
  if (payload.action === 'probe') {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, canWrite: true })).setMimeType(ContentService.MimeType.JSON);
  }
  var sheet = ss.getSheets()[0];
  sheet.clearContents();
  if (payload.rows && payload.rows.length > 0) {
    sheet.getRange(1, 1, payload.rows.length, payload.rows[0].length).setValues(payload.rows);
  }
  var metaSheet = ss.getSheetByName('KINETIC_DATA') || ss.insertSheet('KINETIC_DATA');
  metaSheet.getRange('A1').setValue(JSON.stringify(payload.stateJson));
  return ContentService.createTextOutput(JSON.stringify({ ok: true, updated: new Date().toISOString() })).setMimeType(ContentService.MimeType.JSON);
}
